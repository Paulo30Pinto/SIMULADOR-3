// Classe principal do Simulador
function conectPontaVermelha(posY) {
    //alert(posY);
    if (!this.validarCondicoes("Ponta vermelha")) return;
    this.app.ponta_vermelha_inicial_mc.visible = false;
    this.app.ponta_vermelha_final_mc.visible = true;
    this.app.ponta_vermelha_final_mc.y = posY;
    this.pontaVermelha.conectada = true;
    this.pontaVermelha.posicao = "J1G";
    this.estado = "Ponta vermelha ligada";
    this.atualizarInterface();
    this.verificarMedicao();
}

function conectarPontaPreta(posY) {
    if (!this.validarCondicoes("Ponta preta")) return;
    this.app.ponta_preta_inicial_mc.visible = false;
    this.app.ponta_preta_final_mc.visible = true;
    this.app.ponta_preta_final_mc.y = posY;
    this.pontaPreta.conectada = true;
    this.pontaPreta.posicao = "J1G0";
    this.estado = "Ponta preta ligada";
    this.atualizarInterface();
    this.verificarMedicao();
}

                this.app.placa_electronica_mecanica_btn.visible = false;
                this.app.placa_electronica_electrica_btn.visible = false;
                break;

        }

    }



    //FUNÇÂO PARA LIGAR O SIMULADOR
    ligarSimulador() {
        this.ligado = this.ligado;

        this.app.botao_off_btn.visible = false;
        this.app.botao_on_btn.visible = true;
        this.estado = "Simulador Ligado";
        this.tensaoAlimentacao = 220;

        this.audio.tocarSom("SomBotao");
        this.atualizarInterface();

    }

    //FUNCÇÃO PARA DESLIGAR O SIMULADOR
    desligarSimulador() {
        this.ligado != this.ligado;

        this.app.botao_on_btn.visible = false;
        this.app.botao_off_btn.visible = true;
        this.estado = "Simulador Desligado";
        this.tensaoAlimentacao = 0;

        this.audio.tocarSom("SomBotao");
        this.atualizarInterface();

    }

    //LIGAR E DESLIGAR O SIMULADOR EM UMA UNICA FUNÇÃO
    togglePower() {
        this.ligado = !this.ligado;

        if (this.ligado) {
            this.app.botao_off_btn.visible = false;
            this.app.botao_on_btn.visible = true;
            this.estado = "Simulador Ligado";
            this.tensaoAlimentacao = 220;
        } else {
            this.app.botao_on_btn.visible = false;
            this.app.botao_off_btn.visible = true;
            this.estado = "Simulador Desligado";
            this.tensaoAlimentacao = 0;
        }

        this.audio.tocarSom("SomBotao");
        this.atualizarInterface();
    }

    atualizarInterface() {
        this.app.estado_simulador_txt.text = this.estado;
    }

    conectarPontaVermelha() {
        if (!this.validarCondicoes("Ponta vermelha")) return;

        this.app.ponta_vermelha_inicial_mc.visible = false;
        this.app.ponta_vermelha_final_mc.visible = true;
        this.pontaVermelha.conectada = true;
        this.pontaVermelha.posicao = "J1G";
        this.estado = "Ponta vermelha ligada";
        this.atualizarInterface();
        this.verificarMedicao();
    }

    conectarPontaPreta(posY) {
        if (!this.validarCondicoes("Ponta preta")) return;

        this.app.ponta_preta_inicial_mc.visible = false;
        this.app.ponta_preta_final_mc.visible = true;
        this.app.ponta_preta_final_mc.y = posY;
        this.pontaPreta.conectada = true;
        this.pontaPreta.posicao = "J1G0";
        this.estado = "Ponta preta ligada";
        this.atualizarInterface();
        this.verificarMedicao();
    }

    validarCondicoes(ponta) {
        if (!this.ligado) {
            this.estado = "Ligar Tensão de Alimentação";
            if (ponta === "Ponta vermelha") {
                this.app.ponta_vermelha_inicial_mc.visible = true;
                this.app.ponta_vermelha_final_mc.visible = false;
            } else {
                this.app.ponta_preta_inicial_mc.visible = true;
                this.app.ponta_preta_final_mc.visible = false;
            }
            this.atualizarInterface();
            return false;
        }

        if (this.multimetro.posicaoRoda !== 1) {
            this.estado = "Colocar multimetro em tensao AC";
            if (ponta === "Ponta vermelha") {
                this.app.ponta_vermelha_inicial_mc.visible = true;
                this.app.ponta_vermelha_final_mc.visible = false;
            } else {
                this.app.ponta_preta_inicial_mc.visible = true;
                this.app.ponta_preta_final_mc.visible = false;
            }
            this.atualizarInterface();
            return false;
        }

        return true;
    }

    verificarMedicao() {
        if (this.pontaVermelha.conectada && this.pontaPreta.conectada) {
            if (this.pontaVermelha.posicao === "J1G" && this.pontaPreta.posicao === "J1G0") {
                this.app.display_inferior_txt.text = this.tensaoAlimentacao.toString();
            }
            // Resetar as pontas após a medição
            this.pontaVermelha.conectada = false;
            this.pontaPreta.conectada = false;
        }
    }
}

// Classe para o Multímetro
class Multimetro {
    constructor(app, simulador) {
        this.app = app;
        this.simulador = simulador;
        this.posicaoRoda = 0;
        this.menuAberto = false;

        // Mapeamento de posições e displays
        this.posicoes = [
            { nome: "Desligado", rotacao: 0, displayText: "--------", statusText: "" },
            { nome: "Volts AC", rotacao: 25, displayText: "Volts AC", statusText: "Multímetro ligado para Volts AC" },
            { nome: "Volts DC", rotacao: 50, displayText: "Volts DC", statusText: "Multímetro ligado para Volts DC" },
            { nome: "Volts mV", rotacao: 75, displayText: "Volts mV", statusText: "Multímetro ligado para Volts mV" },
            { nome: "Ohms", rotacao: 100, displayText: "Ohms", statusText: "Multímetro ligado para Ohms" },
            { nome: "Microfarads", rotacao: 125, displayText: "Microfarads", statusText: "Multímetro ligado para Microfarads" },
            { nome: "Anpers", rotacao: 150, displayText: "Anpers", statusText: "Multímetro ligado para Anpers" }
        ];

        this.initEventListeners();
    }

    initEventListeners() {
        // Menu multímetro
        this.app.menu_multimetro_btn.addEventListener("click", this.toggleMenu.bind(this));

        // Roda multímetro
        this.app.roda_multimetro_btn.addEventListener("click", this.girarRoda.bind(this));

        // Opções do menu
        this.app.menu_tensao_ac_btn.addEventListener("click", () => this.selecionarFuncao(1));
        this.app.menu_tensao_dc_btn.addEventListener("click", () => this.selecionarFuncao(2));
        this.app.menu_tensao_ac_milivolts_btn.addEventListener("click", () => this.selecionarFuncao(3));
        this.app.menu_resistencia_btn.addEventListener("click", () => this.selecionarFuncao(4));
        this.app.menu_capacidade_btn.addEventListener("click", () => this.selecionarFuncao(5));
        this.app.menu_corrente_btn.addEventListener("click", () => this.selecionarFuncao(6));
    }

    toggleMenu() {
        this.menuAberto = !this.menuAberto;

        this.app.menu_tensao_ac_btn.visible = this.menuAberto;
        this.app.menu_tensao_dc_btn.visible = this.menuAberto;
        this.app.menu_tensao_ac_milivolts_btn.visible = this.menuAberto;
        this.app.menu_resistencia_btn.visible = this.menuAberto;
        this.app.menu_capacidade_btn.visible = this.menuAberto;
        this.app.menu_corrente_btn.visible = this.menuAberto;

        this.app.menu_ligar_btn.visible = false;
        this.app.menu_desligar_btn.visible = false;

        if (this.menuAberto) {
            this.app.capa_display_mc.visible = false;
        } else {
            this.resetarMultimetro();
        }
    }

    resetarMultimetro() {
        this.posicaoRoda = 0;
        this.app.roda_multimetro_btn.rotation = 0;
        this.app.capa_display_mc.visible = true;
        this.app.display_texto_superior_txt.text = "--------";
    }

    girarRoda() {
        this.posicaoRoda = (this.posicaoRoda + 1) % this.posicoes.length;
        this.atualizarPosicaoRoda();
    }

    selecionarFuncao(posicao) {
        this.posicaoRoda = posicao;
        this.atualizarPosicaoRoda();
        this.simulador.audio.tocarSom("SomBotao");
    }

    atualizarPosicaoRoda() {
        const posicao = this.posicoes[this.posicaoRoda];
        this.app.roda_multimetro_btn.rotation = posicao.rotacao;
        this.app.display_texto_superior_txt.text = posicao.displayText;
        this.app.capa_display_mc.visible = this.posicaoRoda === 0;

        if (posicao.statusText) {
            this.simulador.estado = posicao.statusText;
            this.simulador.atualizarInterface();
        }

        this.simulador.audio.tocarSom("SomBotao");
    }
}
